```rust
pub fn build_system_prompt(state: &AgentState, memory_contents: &[String]) -> String {
    let mut prompt = state.instructions.clone();
    
    // Add relevant facts from memory (top 5)
    if !memory_contents.is_empty() {
        push_str(&mut prompt, "\n\n## Relevant Context");
        for (i, fact) in memory_contents.iter().take(5).enumerate() {
            push_str(&mut prompt, &format!("\n{}. {}", i + 1, fact));
        }
    }
    
    // Add working memory (current task context)
    if let Some(last_user) = state.messages.iter().rev().find(|m| m.role == "user") {
        push_str(&mut prompt, &format!("\n\n## Current Task\n{}", last_user.content));
    }
    
    // Ensure under 4K tokens (rough: chars / 4)
    let max_chars = 16000;
    if prompt.len() > max_chars {
        prompt.truncate(max_chars);
        prompt.push_str("\n\n[Context truncated due to length]");
    }
    
    prompt
}
```
